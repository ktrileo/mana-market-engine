 43@app.get("/api/v1/movers")
 44async def get_market_movers(
 45    direction: str = Query("gainers", pattern="^(gainers|losers)$"),
 46    days: int = Query(1, ge=1, le=30),
 47    limit: int = Query(10, ge=1, le=50)
 48):
 49    """Fetch the top market gainers or losers, leveraging Redis cache."""
 50    
 51    cache_key = f"api:movers:{direction}:d{days}:l{limit}"
 52    
 53    try:
 54        # 1. Check Redis for a cached response
 55        cached_data = await app.state.redis.get(cache_key)
 56        if cached_data:
 57            return Response(content=cached_data, media_type="application/json")
 58            
 59        # 2. Cache Miss: Execute TimescaleDB Query
 60        query = """
 61        WITH current_prices AS (
 62            SELECT DISTINCT ON (card_id) card_id, price_usd AS price_now
 63            FROM fact_card_prices WHERE price_usd IS NOT NULL
 64            ORDER BY card_id, timestamp DESC
 65        ),
 66        past_prices AS (
 67            SELECT DISTINCT ON (card_id) card_id, price_usd AS price_past
 68            FROM fact_card_prices 
 69            WHERE price_usd IS NOT NULL AND timestamp <= NOW() - $1::interval
 70            ORDER BY card_id, timestamp DESC
 71        )
 72        SELECT c.name, c.set_code, c.rarity, cp.price_now, pp.price_past,
 73               (cp.price_now - pp.price_past) AS abs_change,
 74               ROUND(((cp.price_now - pp.price_past) / pp.price_past * 100), 2) AS percent_change
 75        FROM current_prices cp
 76        JOIN past_prices pp ON cp.card_id = pp.card_id
 77        JOIN dim_cards c ON cp.card_id = c.card_id
 78        WHERE pp.price_past >= $2
 79          AND ABS(cp.price_now - pp.price_past) >= $3
 80          AND ABS((cp.price_now - pp.price_past) / pp.price_past * 100) >= $4
 81        """
 82        
 83        if direction == "gainers":
 84            query += " AND (cp.price_now - pp.price_past) > 0 ORDER BY percent_change DESC LIMIT $5;"
 85        else:
 86            query += " AND (cp.price_now - pp.price_past) < 0 ORDER BY percent_change ASC LIMIT $5;"
 87
 88        async with app.state.db_pool.acquire() as conn:
 89            rows = await conn.fetch(
 90                query, 
 91                timedelta(days=days),
 92                MIN_BASE_PRICE,
 93                MIN_ABSOLUTE_CHANGE,
 94                MIN_PERCENT_CHANGE,
 95                limit
 96            )
 97            
 98            # Format and serialize results
 99            response_payload = {
100                "window_days": days,
101                "direction": direction,
102                "cached": False,
103                "results": [dict(row) for row in rows]
104            }
105            
106            # default=str handles asyncpg Decimal conversions natively
107            json_response = json.dumps(response_payload, default=str)
108            
109            # 3. Store in Redis with TTL
110            await app.state.redis.set(cache_key, json_response, ex=CACHE_TTL_SECONDS)
111            
112            return Response(content=json_response, media_type="application/json")
113            
114    except Exception as e:
115        raise HTTPException(status_code=500, detail=str(e))

117@app.get("/api/v1/cards/search")
118async def search_cards(
119    q: str = Query(..., min_length=1, max_length=100, description="Card name search query"),
120    limit: int = Query(10, ge=1, le=50, description="Maximum number of results to return")
121):
122    """Search for cards by name with current market prices and artwork URIs."""
123    clean_query = q.strip()
124    cache_key = f"api:cards:search:{clean_query.lower()}:l{limit}"
125    
126    try:
127        # 1. Check Redis cache
128        cached_data = await app.state.redis.get(cache_key)
129        if cached_data:
130            return Response(content=cached_data, media_type="application/json")
131            
132        # 2. Query TimescaleDB / PostgreSQL
133        query = """
134        SELECT
135            c.card_id,
136            c.oracle_id,
137            c.name,
138            c.set_code,
139            c.collector_number,
140            c.rarity,
141            c.mana_value,
142            COALESCE(
143                c.raw_scryfall_data->'image_uris'->>'normal',
144                c.raw_scryfall_data->'card_faces'->0->'image_uris'->>'normal'
145            ) AS image_uri,
146            COALESCE(
147                c.raw_scryfall_data->'image_uris'->>'small',
148                c.raw_scryfall_data->'card_faces'->0->'image_uris'->>'small'
149            ) AS image_uri_small,
150            c.raw_scryfall_data->>'scryfall_uri' AS scryfall_uri,
151            p.price_usd,
152            p.price_usd_foil,
153            p.price_eur,
154            p.timestamp AS price_updated_at
155        FROM dim_cards c
156        LEFT JOIN LATERAL (
157            SELECT price_usd, price_usd_foil, price_eur, timestamp
158            FROM fact_card_prices fp
159            WHERE fp.card_id = c.card_id
160            ORDER BY fp.timestamp DESC
161            LIMIT 1
162        ) p ON true
163        WHERE c.name ILIKE '%' || $1 || '%'
164        ORDER BY
165            CASE
166                WHEN LOWER(c.name) = LOWER($1) THEN 0
167                WHEN LOWER(c.name) LIKE LOWER($1) || '%' THEN 1
168                ELSE 2
169            END,
170            c.name ASC,
171            c.set_code ASC
172        LIMIT $2;
173        """
174        
175        async with app.state.db_pool.acquire() as conn:
176            rows = await conn.fetch(query, clean_query, limit)
177            
178            response_payload = {
179                "query": clean_query,
180                "count": len(rows),
181                "cached": False,
182                "results": [dict(row) for row in rows]
183            }
184            
185            json_response = json.dumps(response_payload, default=str)
186            await app.state.redis.set(cache_key, json_response, ex=CACHE_TTL_SECONDS)
187            return Response(content=json_response, media_type="application/json")
188            
189    except Exception as e:
190        raise HTTPException(status_code=500, detail=str(e))

192@app.get("/api/v1/cards/{card_id}/history")
193async def get_card_history(
194    card_id: UUID = Path(..., description="Unique card ID (UUID)"),
195    days: int = Query(30, ge=1, le=365, description="Lookback window in days")
196):
197    """Retrieve historical daily price points and financial summary stats for a card."""
198    cache_key = f"api:cards:{card_id}:history:d{days}"
199    
200    try:
201        # 1. Check Redis cache
202        cached_data = await app.state.redis.get(cache_key)
203        if cached_data:
204            return Response(content=cached_data, media_type="application/json")
205            
206        async with app.state.db_pool.acquire() as conn:
207            # 2. Fetch card metadata
208            card_row = await conn.fetchrow("""
209                SELECT
210                    card_id, oracle_id, name, set_code, collector_number, rarity, mana_value,
211                    COALESCE(
212                        raw_scryfall_data->'image_uris'->>'normal',
213                        raw_scryfall_data->'card_faces'->0->'image_uris'->>'normal'
214                    ) AS image_uri,
215                    raw_scryfall_data->>'scryfall_uri' AS scryfall_uri
216                FROM dim_cards
217                WHERE card_id = $1;
218            """, card_id)
219            
220            if not card_row:
221                raise HTTPException(status_code=404, detail=f"Card with ID '{card_id}' not found")
222                
223            # 3. Fetch price history series
224            history_rows = await conn.fetch("""
225                SELECT
226                    timestamp,
227                    vendor,
228                    price_usd,
229                    price_usd_foil,
230                    price_eur
231                FROM fact_card_prices
232                WHERE card_id = $1
233                  AND timestamp >= NOW() - $2::interval
234                ORDER BY timestamp ASC;
235            """, card_id, timedelta(days=days))
236            
237            history_list = [dict(r) for r in history_rows]
238            
239            # 4. Compute financial performance metrics over the requested window
240            valid_usd_prices = [float(h["price_usd"]) for h in history_list if h.get("price_usd") is not None]
241            
242            stats = None
243            if valid_usd_prices:
244                oldest_usd = valid_usd_prices[0]
245                current_usd = valid_usd_prices[-1]
246                abs_diff = round(current_usd - oldest_usd, 2)
247                pct_diff = round((abs_diff / oldest_usd * 100), 2) if oldest_usd > 0 else 0.0
248                stats = {
249                    "current_price_usd": current_usd,
250                    "oldest_price_usd": oldest_usd,
251                    "abs_change_usd": abs_diff,
252                    "percent_change_usd": pct_diff,
253                    "min_price_usd": min(valid_usd_prices),
254                    "max_price_usd": max(valid_usd_prices),
255                    "data_points": len(valid_usd_prices)
256                }
257                
258            response_payload = {
259                "card": dict(card_row),
260                "window_days": days,
261                "stats": stats,
262                "cached": False,
263                "history": history_list
264            }
265            
266            json_response = json.dumps(response_payload, default=str)
267            await app.state.redis.set(cache_key, json_response, ex=CACHE_TTL_SECONDS)
268            return Response(content=json_response, media_type="application/json")
269            
270    except HTTPException:
271        raise
272    except Exception as e:
273        raise HTTPException(status_code=500, detail=str(e))
